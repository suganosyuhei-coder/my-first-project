export function drawEnemies(ctx,width,height,game){
 const state=game.enemies;if(!state)return;const scale=game.viewport.scale,screen=p=>worldToScreen(p,game.flight.position,width,height,scale);
 ctx.save();
 for(const site of game.campaign?.sites??[]){
  const p=screen(site.position),r=site.radius*scale;
  if(p.x+r<0||p.x-r>width||p.y+r<0||p.y-r>height)continue;
  ctx.strokeStyle=site.kind==='home'?'#aa605877':'#a87d6877';ctx.lineWidth=1;ctx.setLineDash([5,8]);ctx.beginPath();ctx.arc(p.x,p.y,r,0,Math.PI*2);ctx.stroke();ctx.setLineDash([]);
  if(site.kind==='home'){
   for(let i=0;i<12;i++){const a=i*Math.PI/6;ctx.strokeRect(p.x+Math.cos(a)*r*.85-5,p.y+Math.sin(a)*r*.85-5,10,10);}
  }else{
   ctx.save();ctx.translate(p.x,p.y);ctx.strokeStyle='#ad8170';ctx.strokeRect(-24*scale,-24*scale,48*scale,48*scale);ctx.strokeRect(-13*scale,-13*scale,26*scale,26*scale);ctx.restore();
  }
  ctx.fillStyle='#d5a18a';ctx.font='10px ui-monospace,monospace';ctx.fillText(site.name,p.x-r*.6,p.y+r+16);
 }
 for(const enemy of state.units){if(enemy.hp<=0||!enemy.detected)continue;const type=ENEMY_TYPES.find(t=>t.id===enemy.type),p=screen(enemy.position),r=type.radius*scale;if(p.x+r<0||p.x-r>width||p.y+r<0||p.y-r>height)continue;ctx.save();ctx.translate(p.x,p.y);ctx.rotate(enemy.heading);ctx.fillStyle='#16202e';ctx.strokeStyle=type.color;ctx.lineWidth=1.5;
  if(type.weakpoint){
   ctx.fillStyle='#332932';ctx.strokeStyle=type.color;ctx.lineWidth=2;
   ctx.beginPath();ctx.moveTo(0,-r);ctx.lineTo(r*.72,-r*.65);ctx.lineTo(r,r*.35);ctx.lineTo(r*.55,r*.85);ctx.lineTo(0,r*.62);ctx.lineTo(-r*.55,r*.85);ctx.lineTo(-r,r*.35);ctx.lineTo(-r*.72,-r*.65);ctx.closePath();ctx.fill();ctx.stroke();
   for(const x of [-.58,.58]){ctx.strokeRect(x*r-r*.16,-r*.3,r*.32,r*.7);ctx.fillStyle='#7d4c50';ctx.fillRect(x*r-r*.1,-r*.23,r*.2,r*.5);}
   ctx.fillStyle='#ffdf84';ctx.beginPath();ctx.arc(0,-r*.72,r*.3,0,Math.PI*2);ctx.fill();ctx.strokeStyle='#fff5cf';ctx.stroke();
  }else if(enemy.type==='miniboss'){
   ctx.fillStyle='#3b2930';ctx.strokeStyle=type.color;ctx.beginPath();for(let i=0;i<8;i++){const a=i*Math.PI/4,x=Math.cos(a)*r,y=Math.sin(a)*r;if(i===0)ctx.moveTo(x,y);else ctx.lineTo(x,y);}ctx.closePath();ctx.fill();ctx.stroke();ctx.strokeRect(-r*.5,-r*.5,r,r);
  }else if(type.supportRange){ctx.strokeStyle='#D6B7F433';ctx.setLineDash([3,8]);ctx.beginPath();ctx.arc(0,0,type.supportRange*scale,0,Math.PI*2);ctx.stroke();ctx.setLineDash([]);ctx.strokeStyle=type.color;ctx.fillRect(-r,-r,r*2,r*2);ctx.strokeRect(-r,-r,r*2,r*2);ctx.strokeRect(-r*.5,-r*.5,r,r);}
  else{ctx.beginPath();ctx.moveTo(0,-r);ctx.lineTo(r,r);ctx.lineTo(0,r*.5);ctx.lineTo(-r,r);ctx.closePath();ctx.fill();ctx.stroke();if(type.shootRange>0)ctx.fillRect(-2,-r-4,4,7);}

 }
 for(const boss of state.units.filter(e=>e.hp>0&&e.warning!==undefined)){
  const type=ENEMY_TYPES.find(t=>t.id===boss.type),core=enemyWeakpoint(boss),from=screen(core.position);ctx.strokeStyle='#ff7777';ctx.lineWidth=1;ctx.setLineDash([6,6]);

 }

  for(let i=0;i<10;i++){const angle=i*2.399+debris.heading,reach=r*(.25+t*1.25),x=p.x+Math.cos(angle)*reach,y=p.y+Math.sin(angle)*reach+t*t*r*.7,size=Math.max(2,r*(i%3===0?.3:.18));ctx.save();ctx.translate(x,y);ctx.rotate(angle+t*(i%2?2:-2));ctx.fillStyle=i%3?'#59616b':type.color;ctx.strokeStyle='#a3abb0';ctx.lineWidth=.6;ctx.beginPath();ctx.moveTo(-size,-size*.4);ctx.lineTo(size*.7,-size);ctx.lineTo(size*.3,size);ctx.closePath();ctx.fill();ctx.stroke();ctx.restore();}ctx.restore();
 }
 for(const shot of state.shots){const p=screen(shot.position);ctx.fillStyle='#D95757';ctx.fillRect(p.x-2,p.y-3,4,6);}
 for(const beam of state.beams){const from=screen(beam.from),to=screen(beam.to);ctx.globalAlpha=Math.max(0,1-beam.age/BIT_DEFENSE.beamSeconds);ctx.strokeStyle='#79D2DF';ctx.lineWidth=4;ctx.beginPath();ctx.moveTo(from.x,from.y);ctx.lineTo(to.x,to.y);ctx.stroke();ctx.strokeStyle='#efffff';ctx.lineWidth=1.3;ctx.stroke();}
 ctx.restore();
}

### ビット3基とビットシュート
- フォースを除く回収用ビットは3基。波動砲の発射に合わせ、最大未満で1基、最大で2基、拡散で3基まで待機ビットを発射する。通常弾では発動しない。
- 各ビットは自身から最寄りの600m以内の敵へ体当たりし、命中後に最寄りの敵を選び直す。同じ敵が最寄りなら繰り返し攻撃する。8回のダメージで帰還し、敵がいなくなった場合も帰還する。
- 暫定値は速度420m/s・1命中18ダメージ・命中間隔0.2秒。帰還速度は自機速度を加算する。回収中・帰還中・攻撃中のビットを新しい攻撃で横取りしない。

### 船体グリッドの損傷表示
- 被弾・岩の衝突などで実際に減ったセル／装備の耐久値を、グリッド上の赤い枠と記号で表示する。耐久35％超の損傷は△、35％以下は！、破壊は×とし、残り耐久％を付近に表示する。
- 複数セル装備は最後に実際の衝撃を受けたセルへ表示する。破壊で部品が消えても×を残す。無傷・外観変化だけの接触・無敵によって拒否された攻撃では表示を増やさず、再挑戦で消す。

### 金属板・ワイヤーによる修復
- 金属板／ワイヤーの回収完了、またはフォースによる直接取得で、最も耐久が低い箇所を最大耐久の8％分、そのグリッド座標の上下を各5％分修復する。上限100％。同じ複数セル装備は一度だけ修復し、横のセルは対象外。素材の搬入は従来どおり加算する。
- 修復セルに約2.2秒間、緑の枠・十字・実回復％を表示する。重要部の耐久も同期する。航行終了後は修復しない。破壊箇所も修復対象だが、新装備と重なる破壊済み旧装備は復活させない。再挑戦でエフェクトをリセットする。

### 破壊報酬による修復・表情の中間コマ
- PAWアーマー撃破では強化パーツ1個の取得に加え、最も損傷した箇所を8％、上下を各5％修復する。直接反映し、回収操作は不要。
- 岩の破壊では約80％で金属板1個を自動取得し、その取得時に最も損傷した箇所を4％、上下を各2.5％修復する。氷・コンテナはこの金属報酬の対象外。既存の強化パーツ抽選は維持し、同時取得できる。同じ岩や敵からの二重報酬・二重修復を防ぐ。
- 上記の修復でも既存の緑の回復表示・耐久上限・重複装備保護を適用する。
- 基本表情から少し不安・驚き・困惑・考え中・目を逸らす・疑問へ切り替わる際、ユーザー提供の中間5コマを各0.1秒ずつ順番に再生し、最優先シートの静止差分に落ち着く。同じ表情でのセリフ変更では再生し直さない。一時停止中はコマ送りを停止し、動きを抑える設定では中間コマを省く。

- 表情の中間コマ追加：基本から動揺・呆れ・決意・寂しげ・目を閉じる・驚き（強）への6種類にも、ユーザー提供の5コマを各0.1秒で一度だけ再生する。既存6種類と合わせて12種類。終了後は最優先シートの静止差分を使用する。
- 拡散レーザーの先端当たり判定は、明るい先端の外周幅に合わせて基準半径の約4.22倍とする。見た目のサイズは維持し、淡い発光や残光全体は当たり判定に含めない。敵・岩・氷に適用し、各レーザーが同じ対象へ繰り返しダメージを与えない判定を維持する。

- 強化パーツの取得時修復：タンク・エンジン・射撃強化装置・パネル・観測装置を取得した際、最も損傷した箇所を8％、上下各4％修復する。ビット帰還・フォース回収・岩氷ドロップ・PAW報酬・レア報酬に適用し、装着できず保管した場合も取得として扱う。PAW報酬の従来8％／5％修復はこの8％／4％に統一し二重修復しない。金属板・ワイヤー単独回収と岩の金属板報酬は、それぞれ既存の修復率を維持する。緑の回復表示も共通で使用する。
- 拡散レーザー全体の幅を従来の1.4倍にする。先端・帯・発光・残光を同じ比率で太くし、先端の当たり判定も1.4倍に拡大する。通常の波動砲・通常弾の幅は維持する。

### パイロットのバイタル
- 差分・セリフ表示の下へVITALゲージを配置する。最大100・初期80。航行中のシミュレーション時間で1秒ごとに1減少し、一時停止・航行終了中は減少しない。
- 氷の破壊時は水1個を自動取得し、バイタルを5回復する。上限100。同じ氷からの二重取得・二重回復を防ぐ。水の取得数を所持品欄へ表示する。
- 実際にバイタルが回復した際は、安心・安堵の表情とゲージの緑色の強調を表示する。被弾・危機などの緊急会話を優先し、回復反応は短時間待機できる。
- バイタル0で航行を終了し、現在のゲームオーバー画面に「パイロットのバイタル切れ」と表示する。船体破壊は発生させず、射撃・ブースト・敵弾・回収を終了する。専用演出は将来対応。再挑戦では80、水の所持数0へリセットする。

### 船体崩壊・低バイタル警告と終了演出
